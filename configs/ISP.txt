# ISP
# Practica 3 - Infraestructura 2 (Jump Server) - Aaron Hernandez - 2025-0800

enable
configure terminal
hostname ISP
no ip domain-lookup
banner motd #ISP - Lab Jump Server - Matricula 2025-0800#
ip routing

interface Ethernet0/0
 description Enlace hacia R-USER e0/0
 ip address 20.25.8.1 255.255.255.252
 no shutdown
exit

interface Ethernet0/1
 description Enlace hacia FG-Server port1
 ip address 20.25.0.1 255.255.255.252
 no shutdown
exit

interface Loopback0
 description Simula Internet
 ip address 8.8.8.8 255.255.255.255
exit

line console 0
 logging synchronous
 exec-timeout 0 0
exit

end
write memory

! (Solo deben aparecer 20.25.8.0/30, 20.25.0.0/30 y 8.8.8.8/32.
!  El ISP no conoce las redes 10.8.0.x, al Jump Server solo se llega por la VPN)
