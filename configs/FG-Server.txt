#Paso 0: acceso de gestion (unico paso por consola)
config system interface
    edit "port3"
        set mode static
        set ip 192.168.128.141 255.255.255.0
        set allowaccess ping https ssh http
    next
end

#Hostname y zona horaria (GUI: System > Settings)
config system global
    set hostname "FG-Server"
    set timezone 77
end

#Interfaces (GUI: Network > Interfaces)
config system interface
    edit "port1"
        set alias "WAN"
        set mode static
        set ip 20.25.0.2 255.255.255.252
        set allowaccess ping
        set role wan
    next
    edit "port2"
        set alias "LAN-JUMP"
        set mode static
        set ip 10.8.0.137 255.255.255.248
        set allowaccess ping
        set role lan
    next
    edit "port4"
        set alias "LAN-WEB"
        set mode static
        set ip 10.8.0.129 255.255.255.248
        set allowaccess ping
        set role lan
    next
end

#Ruta por defecto hacia el ISP (GUI: Network > Static Routes)
config router static
    edit 1
        set gateway 20.25.0.1
        set device "port1"
    next
end

# ============================================================
# OBJETOS (GUI: Policy & Objects > Addresses)
# ============================================================
config firewall address
    edit "Jump-Server"
        set subnet 10.8.0.138 255.255.255.255
    next
    edit "Web-Server"
        set subnet 10.8.0.130 255.255.255.255
    next
end

# ============================================================
# USUARIOS Y GRUPOS (GUI: User & Authentication)
# lmartinez = sin privilegios / agomez = con privilegios
# ============================================================
config user local
    edit "lmartinez"
        set type password
        set passwd <PASSWORD>
    next
    edit "agomez"
        set type password
        set passwd <PASSWORD>
    next
end

config user group
    edit "VPN-Estandar"
        set member "lmartinez"
    next
    edit "VPN-Admin"
        set member "agomez"
    next
end

# ============================================================
# VPN DE ACCESO REMOTO IPsec (GUI: VPN > IPsec Wizard > Remote Access,
# despues "Convert To Custom Tunnel")
# ============================================================

#Objetos que crea el wizard
config firewall address
    edit "VPN-JUMP_range"
        set type iprange
        set start-ip 10.8.0.200
        set end-ip 10.8.0.210
    next
end

config firewall addrgrp
    edit "VPN-JUMP_split"
        set member "Jump-Server"
    next
end

#Phase 1
#(DES/MD5 y DH 5 por la licencia de evaluacion)
#(XAuth sin grupo fijo: "Inherit from policy", cada politica dice que grupo entra)
#(Split tunnel: al cliente solo se le manda la ruta del Jump Server)
config vpn ipsec phase1-interface
    edit "VPN-JUMP"
        set type dynamic
        set interface "port1"
        set ike-version 1
        set mode aggressive
        set peertype any
        set net-device disable
        set mode-cfg enable
        set proposal des-md5
        set dhgrp 5
        set xauthtype auto
        set ipv4-start-ip 10.8.0.200
        set ipv4-end-ip 10.8.0.210
        set dns-mode auto
        set ipv4-split-include "VPN-JUMP_split"
        set psksecret <PRE-SHARED-KEY>
    next
end

#Phase 2
#(Selectores 0.0.0.0/0, normal en acceso remoto. El acceso lo limitan las politicas)
config vpn ipsec phase2-interface
    edit "VPN-JUMP"
        set phase1name "VPN-JUMP"
        set proposal des-md5
        set pfs enable
        set dhgrp 5
        set keepalive enable
        set keylifeseconds 43200
    next
end

# ============================================================
# POLITICAS (GUI: Policy & Objects > Firewall Policy)
# Orden de arriba hacia abajo. Se usan 4 de las 5 que permite la licencia
# ============================================================
config firewall policy
    edit 3
        set name "Bloqueo-SSH-Estandar"
        set srcintf "VPN-JUMP"
        set dstintf "port2"
        set srcaddr "VPN-JUMP_range"
        set dstaddr "Jump-Server"
        set action deny
        set schedule "always"
        set service "SSH"
        set logtraffic all
        set groups "VPN-Estandar"
        set comments "Bloqueo explicito de SSH al usuario sin privilegios"
    next
    edit 2
        set name "Estandar-HTTPS-Jump"
        set srcintf "VPN-JUMP"
        set dstintf "port2"
        set srcaddr "VPN-JUMP_range"
        set dstaddr "Jump-Server"
        set action accept
        set schedule "always"
        set service "HTTPS"
        set logtraffic all
        set groups "VPN-Estandar"
        set comments "Usuario sin privilegios: solo HTTPS al Jump Server"
    next
    edit 1
        set name "Admin-Acceso-Jump"
        set srcintf "VPN-JUMP"
        set dstintf "port2"
        set srcaddr "VPN-JUMP_range"
        set dstaddr "Jump-Server"
        set action accept
        set schedule "always"
        set service "HTTPS" "RDP"
        set logtraffic all
        set groups "VPN-Admin"
        set comments "Usuario con privilegios: HTTPS y RDP al Jump Server"
    next
    edit 4
        set name "Jump-Servicios-Web"
        set srcintf "port2"
        set dstintf "port4"
        set srcaddr "Jump-Server"
        set dstaddr "Web-Server"
        set action accept
        set schedule "always"
        set service "HTTPS" "RDP" "SSH"
        set logtraffic all
        set comments "Jump Server al Web Server: HTTPS, RDP y SSH"
    next
end

#Implicit Deny con log (GUI: clic derecho en Implicit Deny > Edit)
config log setting
    set fwpolicy-implicit-log enable
end
